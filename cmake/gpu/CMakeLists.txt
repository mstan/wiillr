cmake_minimum_required(VERSION 3.25)
project(wiillr_graphics LANGUAGES C CXX)
include("${CMAKE_CURRENT_LIST_DIR}/../CompilerRuntime.cmake")
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_TESTING OFF CACHE BOOL "Build donor tests")
set(WIILLR_CORE_SOURCE_DIR "" CACHE PATH "Pinned gc-wii-core source")
set(GXRUNTIME_DIR "" CACHE PATH "Pinned RecompCore GXRuntime source")
if(NOT EXISTS "${GXRUNTIME_DIR}/graphics/aurora/CMakeLists.txt" OR
   NOT EXISTS "${WIILLR_CORE_SOURCE_DIR}/native/gx/CMakeLists.txt")
  message(FATAL_ERROR "Supply the pinned GXRuntime and gc-wii-core sources")
endif()
# Reuse only the donor dependency graph. No donor game executable is built.
set(GC_WII_GX_WII ON CACHE BOOL "Wii address fields")
set(GCNLLR_RENDER_PIPELINE retail CACHE STRING "Qualified renderer")
set(GCNLLR_VERTEX_PREPARATION planned CACHE STRING "Qualified vertex preparation")
set(GCNLLR_NATIVE_PIPELINE OFF CACHE BOOL "Experimental pipeline")
set(GCNLLR_NATIVE_DRAW_SINK OFF CACHE BOOL "Experimental sink")
set(GCNLLR_ASYNC_COPIES OFF CACHE BOOL "Experimental streamed copies")
set(GCNLLR_RESIDENT_XFB OFF CACHE BOOL "Experimental retained XFB")
set(GCNLLR_GX_COOPERATIVE_VI OFF CACHE BOOL "Experimental VI cooperation")
set(GCNLLR_GX_FIFO_WORKER OFF CACHE BOOL "FIFO worker")
set(GCNLLR_COMPOSED_SCANOUT OFF CACHE BOOL "Experimental composed scanout")
set(GCNLLR_GX_SUBMISSION segmented CACHE STRING "Qualified submission")
set(GCNLLR_NATIVE_OUTPUT_SUBMISSION queued CACHE STRING "Qualified native output")
set(GCNLLR_GX_COMPLETION physical CACHE STRING "Qualified completion")
set(GCNLLR_GPU_BACKEND auto CACHE STRING "Platform GPU backend")
set(AURORA_DAWN_PROVIDER package CACHE STRING "Pinned Dawn dependency")
set(AURORA_DAWN_LINKAGE shared CACHE STRING "Dawn runtime linkage")
set(AURORA_SDL3_PROVIDER system CACHE STRING "Use the same SDL3 as the CPU")
set(AURORA_SDL3_LINKAGE shared CACHE STRING "SDL3 runtime linkage")
add_subdirectory("${GXRUNTIME_DIR}" gxruntime EXCLUDE_FROM_ALL)
include("${WIILLR_CORE_SOURCE_DIR}/native/gx/CMakeLists.txt")
if(WIILLR_RELEASE_BIN_DIR)
  set_target_properties(gcnllr-native-gx PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${WIILLR_RELEASE_BIN_DIR}"
    LIBRARY_OUTPUT_DIRECTORY "${WIILLR_RELEASE_BIN_DIR}")
endif()

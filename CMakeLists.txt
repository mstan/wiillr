    endif()
    file(STRINGS "${WIILLR_PPC_GENERATED_DIR}/generated_abi.h" _wii_chain_abi
      REGEX "#ifndef DOLRECOMP_NATIVE_CHAIN")
    if(NOT _wii_chain_abi)
      message(FATAL_ERROR "Native chaining requires a newly generated catalog with chain support")
    endif()
  endif()
  file(GLOB _wii_ppc_chunks CONFIGURE_DEPENDS "${WIILLR_PPC_GENERATED_DIR}/chunks/*.c")
  if(WIILLR_PPC_REUSE_DIR)
    set(_wii_compatible_reuse ${WIILLR_PPC_REUSE_DIR})
    if(WIILLR_NATIVE_CHAIN_CATALOG)
      # Relative ABI includes are part of each reused source's identity.
      file(SHA256 "${WIILLR_PPC_GENERATED_DIR}/generated_abi.h" _current_abi)
      set(_wii_compatible_reuse)
      foreach(_reuse IN LISTS WIILLR_PPC_REUSE_DIR)
        if(EXISTS "${_reuse}/generated_abi.h")
          file(SHA256 "${_reuse}/generated_abi.h" _previous_abi)
          if(_current_abi STREQUAL _previous_abi)
            list(APPEND _wii_compatible_reuse "${_reuse}")
          endif()
        endif()
      endforeach()
    endif()
    set(_wii_selected_chunks)
    foreach(_chunk IN LISTS _wii_ppc_chunks)
      get_filename_component(_name "${_chunk}" NAME)
      file(SHA256 "${_chunk}" _current_sha)
      foreach(_reuse IN LISTS _wii_compatible_reuse)
        set(_previous "${_reuse}/chunks/${_name}")
        if(EXISTS "${_previous}")
          file(SHA256 "${_previous}" _previous_sha)
          set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_chunk}" "${_previous}")
          if(_current_sha STREQUAL _previous_sha)
            set(_chunk "${_previous}")
            break()
          endif()
        endif()
      endforeach()
      list(APPEND _wii_selected_chunks "${_chunk}")
    endforeach()
    set(_wii_ppc_chunks ${_wii_selected_chunks})
  endif()

      endif()

    endforeach()
  endif()
  if(WIILLR_FP_GUARD_SCOPE)

    endif()

  endif()
  if(WIILLR_IDLE_SKIP)

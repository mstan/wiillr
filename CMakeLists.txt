set(WIILLR_PPC_GENERATED_DIR "" CACHE PATH "Private verified static PPC catalog")
set(WIILLR_PPC_REUSE_DIR "" CACHE STRING "Optional previous immutable PPC catalogs for identical compilation units (semicolon list)")
if(WIILLR_PPC_GENERATED_DIR)
  if(NOT EXISTS "${WIILLR_PPC_GENERATED_DIR}/overlay_module_tables.inc" OR
     NOT EXISTS "${WIILLR_PPC_GENERATED_DIR}/../prepared.json")
    message(FATAL_ERROR "Static PPC catalog needs its verified preparation receipt")
  endif()
  file(SHA256 "${WIILLR_PPC_GENERATED_DIR}/../prepared.json" _wii_ppc_identity)
  file(GLOB _wii_ppc_chunks CONFIGURE_DEPENDS "${WIILLR_PPC_GENERATED_DIR}/chunks/*.c")
  if(WIILLR_PPC_REUSE_DIR)
    set(_wii_selected_chunks)
    foreach(_chunk IN LISTS _wii_ppc_chunks)
      get_filename_component(_name "${_chunk}" NAME)
      file(SHA256 "${_chunk}" _current_sha)
      foreach(_reuse IN LISTS WIILLR_PPC_REUSE_DIR)
        set(_previous "${_reuse}/chunks/${_name}")
        if(EXISTS "${_previous}")
          file(SHA256 "${_previous}" _previous_sha)
          set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_chunk}" "${_previous}")
          if(_current_sha STREQUAL _previous_sha)
            set(_chunk "${_previous}")
            break()
          endif()
        endif()
      endforeach()
      list(APPEND _wii_selected_chunks "${_chunk}")
    endforeach()
    set(_wii_ppc_chunks ${_wii_selected_chunks})
  endif()
  add_library(wii_ppc_catalog OBJECT ${_wii_ppc_chunks})
  if(WIILLR_SCHED STREQUAL "deadline")
    file(STRINGS "${WIILLR_PPC_GENERATED_DIR}/generated_abi.h" _wii_cycle_observer
      REGEX "#define DR_PUBLISH_CYCLES")
    if(NOT _wii_cycle_observer)
      message(FATAL_ERROR "Deadline scheduling requires a catalog generated with cycle observation support")
    endif()
    target_compile_definitions(wii_ppc_catalog PRIVATE DOLRECOMP_OBSERVE_CYCLES=1)
  endif()
  target_link_libraries(wii_ppc_catalog PRIVATE wii_boot)
  add_executable(wiillr-native src/discover.c src/native.c $<TARGET_OBJECTS:wii_ppc_catalog>)
  target_include_directories(wiillr-native PRIVATE "${WIILLR_PPC_GENERATED_DIR}")
  target_compile_definitions(wiillr-native PRIVATE WIILLR_PPC_STATIC=1
    WIILLR_SCHED_DEADLINE=$<STREQUAL:${WIILLR_SCHED},deadline>
    GCN_COST_RING_ENABLED=$<BOOL:${GCN_COST_RING}>
    "WIILLR_PPC_CATALOG_SHA=\"${_wii_ppc_identity}\"")
  target_link_libraries(wiillr-native PRIVATE wii_boot)
  add_executable(wiillr-native-discover src/discover.c src/native.c $<TARGET_OBJECTS:wii_ppc_catalog>)
  target_include_directories(wiillr-native-discover PRIVATE "${WIILLR_PPC_GENERATED_DIR}")
  target_compile_definitions(wiillr-native-discover PRIVATE WIILLR_PPC_STATIC=1 WIILLR_PPC_DISCOVERY=1
    WIILLR_SCHED_DEADLINE=$<STREQUAL:${WIILLR_SCHED},deadline>
    GCN_COST_RING_ENABLED=$<BOOL:${GCN_COST_RING}>
    "WIILLR_PPC_CATALOG_SHA=\"${_wii_ppc_identity}\"")
  target_link_libraries(wiillr-native-discover PRIVATE wii_boot)
endif()

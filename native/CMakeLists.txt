cmake_minimum_required(VERSION 3.16)
project(wiillr_native_layout LANGUAGES CXX)
enable_testing()
set(GC_WII_CORE_DIR "" CACHE PATH "Shared core with the native render API")
if(NOT EXISTS "${GC_WII_CORE_DIR}/include/gcnllr/native_render.h")
  message(FATAL_ERROR "Supply the shared core native rendering API")
endif()
add_library(wii-native-layout menu_picture.cpp menu_text.cpp mm9_compositor.cpp mm9_screen.cpp mm9_overlay.cpp)
set_target_properties(wii-native-layout PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
target_compile_features(wii-native-layout PUBLIC cxx_std_20)
target_include_directories(wii-native-layout PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../include" "${GC_WII_CORE_DIR}/include")
add_executable(test_menu_picture "${CMAKE_CURRENT_SOURCE_DIR}/../tests/test_menu_picture.cpp")
target_link_libraries(test_menu_picture PRIVATE wii-native-layout)
add_test(NAME native_menu_picture_contract COMMAND test_menu_picture)
add_executable(test_mm9_compositor "${CMAKE_CURRENT_SOURCE_DIR}/../tests/test_mm9_compositor.cpp")
target_link_libraries(test_mm9_compositor PRIVATE wii-native-layout)
add_test(NAME native_mm9_compositor_contract COMMAND test_mm9_compositor)
add_executable(test_mm9_screen "${CMAKE_CURRENT_SOURCE_DIR}/../tests/test_mm9_screen.cpp")
target_link_libraries(test_mm9_screen PRIVATE wii-native-layout)
add_test(NAME native_mm9_screen_contract COMMAND test_mm9_screen)
add_executable(test_mm9_overlay "${CMAKE_CURRENT_SOURCE_DIR}/../tests/test_mm9_overlay.cpp")
target_link_libraries(test_mm9_overlay PRIVATE wii-native-layout)
add_test(NAME native_mm9_overlay_contract COMMAND test_mm9_overlay)
